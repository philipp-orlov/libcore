option(CORE_NET_IO_URING "Use the io_uring loop instead of epoll (needs liburing)" OFF)

set(_net_sources src/log.cpp)
if(WIN32)
    list(APPEND _net_sources src/windows/io_loop_iocp.cpp)
elseif(CORE_NET_IO_URING)
    list(APPEND _net_sources src/linux/io_loop_uring.cpp)
else()
    list(APPEND _net_sources src/linux/io_loop_epoll.cpp)
endif()

add_library(core_net STATIC ${_net_sources})
add_library(core::net ALIAS core_net)
target_include_directories(core_net PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>)
target_compile_features(core_net PUBLIC cxx_std_17)
find_package(Threads REQUIRED)
target_link_libraries(core_net PUBLIC core::sched core::sys core::base Threads::Threads)
if(CORE_NET_IO_URING)
    find_path(LIBURING_INCLUDE_DIR liburing.h)
    find_library(LIBURING_LIBRARY NAMES liburing.a uring)
    target_include_directories(core_net SYSTEM PUBLIC ${LIBURING_INCLUDE_DIR})
    target_link_libraries(core_net PUBLIC ${LIBURING_LIBRARY})
endif()
if(WIN32)
    target_link_libraries(core_net PUBLIC ws2_32 mswsock)
endif()
core_target_defaults(core_net)

if(CORE_BUILD_TESTS)
    add_executable(net_test_loop tests/test_loop.cpp)
    target_link_libraries(net_test_loop PRIVATE core::net)
    target_include_directories(net_test_loop PRIVATE ${CMAKE_SOURCE_DIR}/libbase/tests)
    core_target_defaults(net_test_loop)
    add_test(NAME net_loop COMMAND net_test_loop)
endif()
